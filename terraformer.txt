interface IBinarySerializable
{
    + void Serialize(writer: BinaryWriter, message: IMessageService);
    + void Deserialize(reader: BinaryReader, message: IMessageService);
}

class World implements IBinarySerializable
{
    + OnDrawChunk: event Action<Vector3Int, Dictionary<Vector3Int, HashSet<Vector3Int>>>
    + OnNewChanges: event Action<HashSet<Vector3Int>, HashSet<Vector3Int>>
    + OnClear: event Action
    
    + void Add(blockPos: Vector3Int)
    + void Remove(blockPos: Vector3Int)
    + void Clear()
    + void RememberTheFollowing() 
    + void Flush()
    + HashSet<Vector3Int> GetAllBlocks()
    + Dictionary<Vector3Int, HashSet<Vector3Int>> GetChunks()
    + bool Has(blockPos: Vector3Int)
}

class StateBehaviour implements IState
{
    + OnYield: event Action<IState>

    + virtual void Enter()
    + virtual void Exit()
    + virtual void OnFrame()
    + virtual void OnTick()
}

class SelectionInformation
{
    + a: Vector3Int
    + b: Vector3Int
}

class BrushManager extends StateBehaviour
{
    + OnNewBrushSelected: event Action<int>
    
    + void Setup(world: World, eyes: Transform)
    + void AddSelection(a: Vector3Int, b: Vector3Int)
    + void RemoveSelection(a: Vector3Int, b: Vector3Int)
    + override void OnFrame()
}

interface IBrushable 
{
    + void Add(a: Vector3Int, b: Vector3Int)
    + void Remove(a: Vector3Int, b: Vector3Int)
}

class Terraformer extends StateBehaviour
{
    + OnPreview: event Action<SelectionInformation>
    + OnEditSelection: event Action<Vector3Int, Vector3Int>

    + void Setup(world: World)
    + override void OnTick()
}

class Fill implements IBrushable
